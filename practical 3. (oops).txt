{

for(i=0;i<n;i++)
{
scanf("%d",&a[i]);
}

scanf("%d",&key);
printf("menu\n");
printf("\n1.Linear search\n2.binary search\n");

switch(ch)
{
case 1: for(i=0;i<n;i++)
if(a[i]==key)
{
flag=1;
break;
}
case 2: for(lb=0,ub=n-1;lb<=ub;)
{
mid=(lb+ub)/2;
if(key==a[mid])
{
flag=1;
break;
}
else if(key<a[mid])
ub=mid-1;
else
lb=mid+1;
}
break;
default:exit(0);
}
if(flag==1)
printf("Search is successful");
else
printf("search is not successful\n");
getch();
}
